cmake_minimum_required(VERSION 3.8)
project(odin_imu_adapter)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED libusb-1.0)
add_executable(imu_only src/imu_only.cpp)
target_include_directories(imu_only PRIVATE vendor ${LIBUSB_INCLUDE_DIRS})
target_link_libraries(imu_only ${CMAKE_CURRENT_SOURCE_DIR}/vendor/liblydHostApi_arm.a ${LIBUSB_LIBRARIES} OpenSSL::SSL OpenSSL::Crypto pthread rt dl)
ament_target_dependencies(imu_only rclcpp sensor_msgs std_msgs)
install(TARGETS imu_only DESTINATION lib/${PROJECT_NAME})
install(FILES vendor/LICENSE vendor/PROVENANCE.json DESTINATION share/${PROJECT_NAME}/vendor)
ament_package()
